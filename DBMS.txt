/*========================================================= 
CASE STUDY: STUDENT COURSE MANAGEMENT SYSTEM 
Normaliza on: 1NF -> 2NF -> 3NF -> BCNF 
=========================================================*/ 
/*--------------------------------------------------------- 
STEP 1: ORIGINAL RELATION 
R(StudentID, StudentName, DeptID, DeptName, 
CourseID, CourseName, Instructor, Marks) 
Func onal Dependencies: 
StudentID -> StudentName, DeptID 
DeptID -> DeptName 
CourseID -> CourseName, Instructor 
(StudentID, CourseID) -> Marks 
Candidate Key: 
  (StudentID, CourseID) ---------------------------------------------------------*/ 
 
 
CREATE TABLE R ( 
    StudentID INT, 
    StudentName VARCHAR(50), 
    DeptID INT, 
    DeptName VARCHAR(50), 
    CourseID INT, 
    CourseName VARCHAR(50), 
    Instructor VARCHAR(50), 
    Marks INT, 
 
    PRIMARY KEY (StudentID, CourseID) 
);

R(StudentID, StudentName, DeptID, DeptName,

SELECT * FROM R; 
/*--------------------------------------------------------- 
STEP 4: 2NF 
Remove par al dependencies: 
StudentID -> StudentName, DeptID, DeptName 
CourseID -> CourseName, Instructor 
(StudentID, CourseID) -> Marks ---------------------------------------------------------*/ 
 
 
CREATE TABLE Student_2NF ( 
    StudentID INT PRIMARY KEY, 
    StudentName VARCHAR(50), 
    DeptID INT, 
    DeptName VARCHAR(50) 
); 

=================================================================================

CREATE TABLE Course_2NF ( 
    CourseID INT PRIMARY KEY, 
    CourseName VARCHAR(50), 
    Instructor VARCHAR(50) 
); 

=============================================================================

CREATE TABLE Enrollment_2NF ( 
    StudentID INT, 
    CourseID INT, 
    Marks INT, 
 
    PRIMARY KEY (StudentID, CourseID) 
); 
 
 
/*--------------------------------------------------------- 
  STEP 5: 3NF 
 
  Remove transi ve dependency: 
 
  StudentID -> DeptID 
  DeptID -> DeptName 
 
  Therefore, create a separate Department table. ---------------------------------------------------------*/ 
 
 
CREATE TABLE Department ( 
    DeptID INT PRIMARY KEY, 
    DeptName VARCHAR(50) 
); 

=============================================================================

CREATE TABLE Student ( 
    StudentID INT PRIMARY KEY, 
    StudentName VARCHAR(50), 
    DeptID INT, 
 
    FOREIGN KEY (DeptID) 
        REFERENCES Department(DeptID) 
); 

==========================================================================

CREATE TABLE Course ( 
    CourseID INT PRIMARY KEY, 
    CourseName VARCHAR(50), 
    Instructor VARCHAR(50) 
); 

============================================================================================

CREATE TABLE Enrollment ( 
    StudentID INT, 
    CourseID INT, 
    Marks INT, 
 
    PRIMARY KEY (StudentID, CourseID), 
FOREIGN KEY (StudentID) 
REFERENCES Student(StudentID), 
FOREIGN KEY (CourseID) 
REFERENCES Course(CourseID) 
);

    D.DeptName,

v_marks NUMBER := 105; 
invalid_marks EXCEPTION; 
BEGIN -- Check condi on 
    IF v_marks > 100 OR v_marks < 0 THEN 
        RAISE invalid_marks; 
    END IF; 
 
    DBMS_OUTPUT.PUT_LINE('Valid Marks: ' || v_marks); 
 
EXCEPTION 
    WHEN invalid_marks THEN 
        DBMS_OUTPUT.PUT_LINE('Error: Marks should be between 0 and 100'); 
END;

2. User-Defined Excep on: Student Not Found 
DECLARE 
    v_student_id NUMBER := 999; 
    v_name VARCHAR2(50); 
    student_not_found EXCEPTION; 
 
BEGIN 
    SELECT Student_Name 
    INTO v_name 
    FROM Student 
    WHERE Student_ID = v_student_id; 
 
    DBMS_OUTPUT.PUT_LINE('Student Found: ' || v_name); 
 
EXCEPTION 
    WHEN NO_DATA_FOUND THEN 
        RAISE student_not_found; 
 
    WHEN student_not_found THEN 
        DBMS_OUTPUT.PUT_LINE('Error: Student ID does not exist'); 
END;
